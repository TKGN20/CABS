cmake_minimum_required(VERSION 3.16)
project(CABSExperiment LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(OpenMP REQUIRED)
find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
add_executable(cabs_fixed_queries fixed_queries.cpp)
target_include_directories(cabs_fixed_queries PRIVATE ../include)
target_link_libraries(cabs_fixed_queries PRIVATE OpenMP::OpenMP_CXX)
target_compile_options(cabs_fixed_queries PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
option(CABS_SANITIZE "Enable sanitizers" OFF)
if(CABS_SANITIZE)
  target_compile_options(cabs_fixed_queries PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(cabs_fixed_queries PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
add_test(NAME experiment_protocol COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_protocol.py --binary $<TARGET_FILE:cabs_fixed_queries>)
add_test(NAME baseline_oracle COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/baselines/test_oracle.py)
add_test(NAME saved_baseline_receipts COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/delivery/verify_receipts.py)
add_executable(independent_exact_gt baselines/exact_gt.cpp)
target_compile_options(independent_exact_gt PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
if(CABS_SANITIZE)
  target_compile_options(independent_exact_gt PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(independent_exact_gt PRIVATE -fsanitize=address,undefined)
endif()
add_test(NAME independent_exact_ground_truth COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/baselines/test_exact_gt.py --binary $<TARGET_FILE:independent_exact_gt>)
add_test(NAME isolated_ground_truth_protocol COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/baselines/test_prepare_gt.py --binary $<TARGET_FILE:independent_exact_gt>)
add_test(NAME common_method_record_audit COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/baselines/test_audit_run.py --gt-binary $<TARGET_FILE:independent_exact_gt> --cabs-binary $<TARGET_FILE:cabs_fixed_queries>)
add_test(NAME common_cost_component_record COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/baselines/test_cost_record.py --cabs-binary $<TARGET_FILE:cabs_fixed_queries>)
