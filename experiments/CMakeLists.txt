cmake_minimum_required(VERSION 3.16)
project(CABSExperiment LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(OpenMP REQUIRED)
find_package(Python3 3.8 REQUIRED COMPONENTS Interpreter)
add_executable(cabs_fixed_queries fixed_queries.cpp)
target_include_directories(cabs_fixed_queries PRIVATE ../include)
target_link_libraries(cabs_fixed_queries PRIVATE OpenMP::OpenMP_CXX)
target_compile_options(cabs_fixed_queries PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
option(CABS_SANITIZE "Enable sanitizers" OFF)
if(CABS_SANITIZE)
  target_compile_options(cabs_fixed_queries PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(cabs_fixed_queries PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
add_test(NAME experiment_protocol COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_protocol.py --binary $<TARGET_FILE:cabs_fixed_queries>)
