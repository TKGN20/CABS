cmake_minimum_required(VERSION 3.16)
project(CABS VERSION 0.1.0 LANGUAGES CXX)
if(NOT UNIX)
  message(FATAL_ERROR "CABS currently uses POSIX mmap. Build on Linux or Linux in WSL.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenMP REQUIRED)
find_package(Python3 3.8 REQUIRED COMPONENTS Interpreter)
add_executable(cabs src/main.cpp)
target_include_directories(cabs PRIVATE include)
target_link_libraries(cabs PRIVATE OpenMP::OpenMP_CXX)
target_compile_options(cabs PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
option(CABS_SANITIZE "Build with AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
if(CABS_SANITIZE)
  target_compile_options(cabs PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(cabs PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
add_test(NAME semantic COMMAND cabs selftest)
add_test(NAME end_to_end COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/smoke_test.py --binary $<TARGET_FILE:cabs>)
add_test(NAME reporting_validation COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_reporting.py)

add_executable(cabs_mechanism_contract tests/mechanism_contract.cpp)
target_include_directories(cabs_mechanism_contract PRIVATE include)
target_link_libraries(cabs_mechanism_contract PRIVATE OpenMP::OpenMP_CXX)
target_compile_options(cabs_mechanism_contract PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
if(CABS_SANITIZE)
  target_compile_options(cabs_mechanism_contract PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(cabs_mechanism_contract PRIVATE -fsanitize=address,undefined)
endif()
add_test(NAME mechanism_contract COMMAND cabs_mechanism_contract)

add_executable(cabs_scoring_pair EXCLUDE_FROM_ALL benchmarks/scoring_pair.cpp)
target_include_directories(cabs_scoring_pair PRIVATE include)
target_link_libraries(cabs_scoring_pair PRIVATE OpenMP::OpenMP_CXX)
target_compile_options(cabs_scoring_pair PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)

add_executable(cabs_order_pair EXCLUDE_FROM_ALL benchmarks/order_pair.cpp)
target_include_directories(cabs_order_pair PRIVATE include)
target_link_libraries(cabs_order_pair PRIVATE OpenMP::OpenMP_CXX)
target_compile_options(cabs_order_pair PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
